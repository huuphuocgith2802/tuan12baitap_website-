// 1. Hàm kiểm tra và gán thuộc tính ketQua theo yêu cầu Bài 4
function kiemTraKetQua(sinhVien) {
  // Logic kiểm tra điều kiện điểm
  if (sinhVien.diemSo >= 5.0) {
    sinhVien.ketQua = "Đậu";
    sinhVien.xepLoai = sinhVien.diemSo >= 8.5 ? "Giỏi" : (sinhVien.diemSo >= 6.5 ? "Khá" : "Trung bình");
  } else {
    sinhVien.ketQua = "Rớt";
    sinhVien.xepLoai = "Yếu";
  }
  return sinhVien;
}
