Bài 1: Danh bạ điện thoại thông minh (Phonebook)

// 1. Mảng các Object danh bạ
const danhBa = [
  { id: 1, ten: "Huỳnh Hữu Phước", sdt: "0912.345.678", tag: "Trường học" },
  { id: 2, ten: "Nguyễn Thị Mai Lan", sdt: "0987.654.321", tag: "Bạn bè" },
  ...
];

// 2. Hàm tìm kiếm số điện thoại khi nhập tên (dùng filter & toLowerCase)
function timSoDienThoai(tuKhoa) {
  const keyword = tuKhoa.toLowerCase().trim();
  return danhBa.filter(lienHe => 
    lienHe.ten.toLowerCase().includes(keyword)
  );
}

// 3. Hàm tìm kiếm chính xác 1 người (dùng find)
function timMotNguoi(hoTen) {
  return danhBa.find(lienHe => lienHe.ten.toLowerCase() === hoTen.toLowerCase());
}
